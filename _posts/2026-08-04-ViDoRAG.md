### 修改 LLM 模型和加载器

官方代码当前仍把示例模型写成 qwen-vl-max，但本地 LLM 类只有模型名包含 Qwen2.5-VL 时才会加载本地模型；同时加载器硬编码了 flash_attention_2。因此这两处必须修改。

```bash
python - <<'PY'
from pathlib import Path
import shutil


def backup(path: Path) -> None:
    backup_path = path.with_suffix(path.suffix + ".bak")
    if not backup_path.exists():
        shutil.copy2(path, backup_path)
        print(f"[BACKUP] {backup_path}")


def replace_required(path: Path, old: str, new: str) -> None:
    text = path.read_text(encoding="utf-8")

    if new in text:
        print(f"[SKIP] {path}: 已修改")
        return

    if old not in text:
        raise RuntimeError(
            f"\n未在 {path} 中找到目标内容：\n{old}\n"
            "请不要继续运行，先检查文件内容。"
        )

    path.write_text(text.replace(old, new), encoding="utf-8")
    print(f"[OK] {path}")


llm_path = Path("llms/llm.py")
agent_path = Path("vidorag_agents.py")

backup(llm_path)
backup(agent_path)

replace_required(
    llm_path,
    'model_name, torch_dtype="auto", device_map="auto",attn_implementation="flash_attention_2"',
    '''model_name,
            torch_dtype=torch.bfloat16,
            device_map="auto",
            attn_implementation="sdpa",
            low_cpu_mem_usage=True,
        )'''.rstrip(")")
)

replace_required(
    agent_path,
    "vlm = LLM('qwen-vl-max')",
    "vlm = LLM('Qwen/Qwen2.5-VL-7B-Instruct')"
)

print("\n修改完成。")
PY
```

### 创建支持断点续跑的 PDF 渲染脚本

官方脚本会处理 ViDoSeek 和 SlideVQA，使用 pdf2image.convert_from_path 将每个 PDF 页面保存为 文档名_页码.jpg。

```bash
# 创建只处理 ViDoSeek、并支持跳过完整 PDF 的版本：
cat > scripts/render_vidoseek.py <<'PY'
from pathlib import Path

from pdf2image import convert_from_path, pdfinfo_from_path
from tqdm import tqdm


ROOT = Path("data/ViDoSeek")
PDF_DIR = ROOT / "pdf"
IMG_DIR = ROOT / "img"

DPI = 200
THREAD_COUNT = 4
JPEG_QUALITY = 95


def get_page_count(pdf_path: Path) -> int:
    info = pdfinfo_from_path(str(pdf_path))
    return int(info["Pages"])


def is_complete(pdf_path: Path, page_count: int) -> bool:
    return all(
        (IMG_DIR / f"{pdf_path.stem}_{page}.jpg").exists()
        for page in range(1, page_count + 1)
    )


def render_pdf(pdf_path: Path) -> int:
    page_count = get_page_count(pdf_path)

    if is_complete(pdf_path, page_count):
        return page_count

    images = convert_from_path(
        str(pdf_path),
        dpi=DPI,
        thread_count=THREAD_COUNT,
        fmt="jpeg",
    )

    if len(images) != page_count:
        raise RuntimeError(
            f"{pdf_path.name}: 预计 {page_count} 页，"
            f"实际生成 {len(images)} 页"
        )

    for page, image in enumerate(images, start=1):
        output = IMG_DIR / f"{pdf_path.stem}_{page}.jpg"

        if output.exists():
            continue

        image.convert("RGB").save(
            output,
            format="JPEG",
            quality=JPEG_QUALITY,
        )

    return page_count


def main() -> None:
    IMG_DIR.mkdir(parents=True, exist_ok=True)

    pdfs = sorted(PDF_DIR.glob("*.pdf"))
    failed = []
    total_pages = 0

    progress = tqdm(pdfs, desc="Rendering ViDoSeek")

    for pdf in progress:
        try:
            pages = render_pdf(pdf)
            total_pages += pages
            progress.set_postfix(
                pages=total_pages,
                file=pdf.name[:24],
            )
        except Exception as exc:
            failed.append((pdf.name, repr(exc)))
            tqdm.write(f"[FAILED] {pdf.name}: {exc}")

    print("PDF数量：", len(pdfs))
    print("页面总数：", total_pages)
    print("失败PDF数量：", len(failed))

    if failed:
        for name, error in failed:
            print(name, error)
        raise SystemExit(1)


if __name__ == "__main__":
    main()
PY
```

### 运行全量动态混合检索

```bash
cat > scripts/run_vidoseek_retrieval.py <<'PY'
import argparse
import json
import os
from pathlib import Path

from tqdm import tqdm

from search_engine import HybridSearchEngine


def get_image_path(node: dict) -> str:
    data = node["node"]

    path = data.get("image_path")
    if path:
        return path

    metadata = data.get("metadata", {})

    path = (
        metadata.get("file_name")
        or metadata.get("filename")
        or metadata.get("file_path")
    )

    if path is None:
        raise RuntimeError("检索节点中没有页面路径")

    name = os.path.basename(path).replace(".txt", ".jpg")
    return os.path.join("data/ViDoSeek/img", name)


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--vl-model", required=True)
    parser.add_argument("--text-model", required=True)
    parser.add_argument(
        "--output",
        default="data/ViDoSeek/results/retrieval_cache.jsonl",
    )
    args = parser.parse_args()

    annotation = Path("data/ViDoSeek/rag_dataset.json")
    output = Path(args.output)
    output.parent.mkdir(parents=True, exist_ok=True)

    with annotation.open("r", encoding="utf-8") as f:
        examples = json.load(f)["examples"]

    finished = set()

    if output.exists():
        with output.open("r", encoding="utf-8") as f:
            for line in f:
                if line.strip():
                    finished.add(json.loads(line)["uid"])

    examples = [
        item for item in examples
        if item["uid"] not in finished
    ]

    engine = HybridSearchEngine(
        dataset="ViDoSeek",
        embed_model_name_vl=args.vl_model,
        embed_model_name_text=args.text_model,
        topk=10,
        gmm=True,
    )

    for sample in tqdm(examples, desc="Hybrid retrieval"):
        result = engine.search(sample["query"])

        candidates = [
            get_image_path(node)
            for node in result["source_nodes"]
        ]

        record = dict(sample)
        record["recall_results"] = result
        record["candidate_images"] = candidates

        with output.open("a", encoding="utf-8") as f:
            f.write(
                json.dumps(record, ensure_ascii=False)
                + "\n"
            )

    print("检索缓存：", output)


if __name__ == "__main__":
    main()
PY
```

### 运行全部 ViDoRAG 多智能体生成

```bash
# 创建脚本
cat > scripts/run_vidorag_generation.py <<'PY'
import argparse
import json
from pathlib import Path

import torch
from tqdm import tqdm

from llms.llm import LLM
from vidorag_agents import ViDoRAG_Agents


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--model", required=True)
    parser.add_argument(
        "--retrieval-cache",
        default=(
            "data/ViDoSeek/results/"
            "retrieval_cache.jsonl"
        ),
    )
    parser.add_argument(
        "--output",
        default=(
            "data/ViDoSeek/results/"
            "vidorag_qwen25vl7b.jsonl"
        ),
    )
    args = parser.parse_args()

    cache_path = Path(args.retrieval_cache)
    output_path = Path(args.output)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    records = []

    with cache_path.open("r", encoding="utf-8") as f:
        for line in f:
            if line.strip():
                records.append(json.loads(line))

    finished = set()

    if output_path.exists():
        with output_path.open("r", encoding="utf-8") as f:
            for line in f:
                if line.strip():
                    finished.add(json.loads(line)["uid"])

    records = [
        item for item in records
        if item["uid"] not in finished
    ]

    vlm = LLM(args.model)
    agents = ViDoRAG_Agents(vlm)

    for sample in tqdm(records, desc="ViDoRAG generation"):
        candidates = [
            path
            for path in sample["candidate_images"]
            if Path(path).exists()
        ]

        output = dict(sample)

        try:
            answer = agents.run_agent(
                query=sample["query"],
                images_path=candidates,
            )
            output["response"] = answer
            output["generation_status"] = "ok"
        except Exception as exc:
            output["response"] = None
            output["generation_status"] = "failed"
            output["generation_error"] = repr(exc)

        with output_path.open("a", encoding="utf-8") as f:
            f.write(
                json.dumps(output, ensure_ascii=False)
                + "\n"
            )

        torch.cuda.empty_cache()

    print("生成结果：", output_path)


if __name__ == "__main__":
    main()
PY
```

### 计算论文对应的检索指标和本地答案指标

```bash
cat > scripts/evaluate_vidorag_results.py <<'PY'
import json
import re
import string
from collections import Counter
from pathlib import Path


RESULT_FILE = Path(
    "data/ViDoSeek/results/"
    "vidorag_qwen25vl7b.jsonl"
)


def normalize(text) -> str:
    text = "" if text is None else str(text)
    text = text.lower()
    text = text.translate(
        str.maketrans("", "", string.punctuation)
    )
    text = re.sub(r"\b(a|an|the)\b", " ", text)
    text = " ".join(text.split())
    return text


def token_f1(prediction, reference) -> float:
    pred_tokens = normalize(prediction).split()
    ref_tokens = normalize(reference).split()

    if not pred_tokens and not ref_tokens:
        return 1.0

    if not pred_tokens or not ref_tokens:
        return 0.0

    common = Counter(pred_tokens) & Counter(ref_tokens)
    overlap = sum(common.values())

    if overlap == 0:
        return 0.0

    precision = overlap / len(pred_tokens)
    recall = overlap / len(ref_tokens)

    return 2 * precision * recall / (precision + recall)


def retrieved_ids(sample):
    result = []

    for node in sample["recall_results"]["source_nodes"]:
        data = node["node"]
        metadata = data.get("metadata", {})

        path = (
            data.get("image_path")
            or metadata.get("file_name")
            or metadata.get("filename")
            or metadata.get("file_path")
        )

        if path is None:
            continue

        name = Path(path).name
        result.append(Path(name).stem)

    return result


def relevant_ids(sample):
    file_stem = Path(
        sample["meta_info"]["file_name"]
    ).stem

    return [
        f"{file_stem}_{page}"
        for page in sample["meta_info"]["reference_page"]
    ]


def recall_at_k(samples, k):
    values = []

    for sample in samples:
        relevant = set(relevant_ids(sample))
        retrieved = set(retrieved_ids(sample)[:k])

        values.append(
            len(relevant & retrieved) / len(relevant)
        )

    return sum(values) / len(values)


def mrr_at_k(samples, k):
    values = []

    for sample in samples:
        relevant = set(relevant_ids(sample))
        rank_score = 0.0

        for rank, item in enumerate(
            retrieved_ids(sample)[:k],
            start=1,
        ):
            if item in relevant:
                rank_score = 1.0 / rank
                break

        values.append(rank_score)

    return sum(values) / len(values)


def main():
    samples = []

    with RESULT_FILE.open("r", encoding="utf-8") as f:
        for line in f:
            if line.strip():
                samples.append(json.loads(line))

    valid_answers = [
        sample for sample in samples
        if sample.get("response") is not None
    ]

    em_values = [
        float(
            normalize(sample["response"])
            == normalize(sample["reference_answer"])
        )
        for sample in valid_answers
    ]

    f1_values = [
        token_f1(
            sample["response"],
            sample["reference_answer"],
        )
        for sample in valid_answers
    ]

    metrics = {
        "sample_count": len(samples),
        "valid_answer_count": len(valid_answers),
        "generation_success_rate": (
            len(valid_answers) / len(samples)
            if samples else 0.0
        ),
        "Recall@1": recall_at_k(samples, 1),
        "Recall@3": recall_at_k(samples, 3),
        "Recall@5": recall_at_k(samples, 5),
        "Recall@10": recall_at_k(samples, 10),
        "MRR@5": mrr_at_k(samples, 5),
        "ExactMatch": (
            sum(em_values) / len(em_values)
            if em_values else 0.0
        ),
        "TokenF1": (
            sum(f1_values) / len(f1_values)
            if f1_values else 0.0
        ),
    }

    output = RESULT_FILE.with_name(
        RESULT_FILE.stem + "_metrics.json"
    )

    with output.open("w", encoding="utf-8") as f:
        json.dump(
            metrics,
            f,
            ensure_ascii=False,
            indent=2,
        )

    print(json.dumps(metrics, ensure_ascii=False, indent=2))
    print("评测结果：", output)


if __name__ == "__main__":
    main()
PY

python scripts/evaluate_vidorag_results.py
```
